\begin{frame}{Neutron stars and fundamental physics}

  Neutron star mergers, especially when observed as multimessenger events (e.g., GW170817), probe
  \vspace{3mm}

\begin{frame}{\textsc{jester} and sequential Monte Carlo (SMC)}
  \def\x{4mm}
  \def\y{3mm}

  \begin{itemize}
    \item \textsc{jester} \href{https://github.com/nuclear-multimessenger-astronomy/jester}{\textcolor{black}{\faGithub}}~\cite{Wouters:2025zju}: used by LVK and ET

    \vspace{\y}
    
    \item From $N$ events $d_i$ to equation of state (EOS) posterior

    \vspace{-2mm}

    \begin{equation*}
      \mathcal{P}(\theta_{\rm{EOS}} | \{d_1, \dots, d_N\}) \propto \mathcal{L}(\{d_1, \dots, d_N\} | \theta_{\rm{EOS}}) \times \pi(\theta_{\rm{EOS}})
    \end{equation*}

    \vspace{\y}
    
    \item \textbf{\smcpurple{Sequential Monte Carlo}}: particles from prior $\rightarrow$ posterior through a sequence of intermediate densities $p_t$
  \end{itemize}

  \vfill

  \begin{figure}
    \centering
    \includegraphics[width=\textwidth]{Figures/smc_steps.pdf}
  \end{figure}

  \vfill

\end{frame}


\begin{frame}{Likelihood vs.\ data tempering}
  \def\x{5mm}
  \def\y{2mm}
  \def\m{-2mm}

  \begin{enumerate}
    \item \textit{Likelihood tempering}:

    \vspace{\m}
    
    \begin{equation*}
      p_t(\theta) \propto \mathcal{L}(\{d_1, \dots, d_N\} | \theta)^{\red{\beta_t}} \pi(\theta) \, \quad \beta_t: 0 \rightarrow 1
    \end{equation*}

    \vspace{\y}
    
    \item \textit{Data tempering}: 

    \vspace{\m}
    
    \begin{equation*}
      p_t(\theta) \propto \mathcal{L}(\{d_1, \dots, d_{\red{t}}\} | \theta) \pi(\theta) \, \quad t: 1 \rightarrow N
    \end{equation*}
  \end{enumerate}

  \vspace{3mm}

  \begin{figure}
    \centering
    \includegraphics[width=0.95\textwidth]{Figures/smc_temperings.pdf}
  \end{figure}

\end{frame}
